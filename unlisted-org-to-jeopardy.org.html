        const gridElement = document.getElementById('quiz-grid');
        const quizContainer = document.getElementById('quiz-container');
        const questionElement = document.getElementById('question');
        const answerElement = document.getElementById('answer');
        const showAnswerButton = document.getElementById('show-answer-btn');
        const mainScreenButton = document.getElementById('main-screen-btn');
        const resetButton = document.getElementById('reset-btn');

        function loadGrid() {
            categories.forEach(category => {
                const header = document.createElement('div');
                header.textContent = category.category;
                header.className = 'header';
                gridElement.appendChild(header);

                category.questions.forEach(question => {
                    const questionDiv = document.createElement('div');
                    questionDiv.textContent = question.difficulty;
                    questionDiv.onclick = () => {
                        if (!questionDiv.classList.contains('disabled')) showQuestion(question, questionDiv);
                    };
                    gridElement.appendChild(questionDiv);
                });
            });
        }

        function showQuestion(question, questionDiv) {
            gridElement.style.display = 'none';
            quizContainer.style.display = 'block';
            questionElement.textContent = question.question;
            answerElement.textContent = `Answer: ${question.answer}`;
            answerElement.style.display = 'none';
            showAnswerButton.style.display = 'block';
            mainScreenButton.style.display = 'none';                                                     resetButton.style.display = 'none';

            showAnswerButton.onclick = () => {
                answerElement.style.display = 'block';
                showAnswerButton.style.display = 'none';
                mainScreenButton.style.display = 'block';
            };

            mainScreenButton.onclick = () => {
                quizContainer.style.display = 'none';
                gridElement.style.display = 'grid';
                questionDiv.classList.add('disabled');
                resetButton.style.display = 'block'; // Show reset button on the main screen
                questionDiv.classList.add('disabled'); // Disable clicked question
            };
        }

        resetButton.onclick = () => {
            document.querySelectorAll('.quiz-grid div').forEach(div => div.classList.remove('disabled'));
        };

        loadGrid();
