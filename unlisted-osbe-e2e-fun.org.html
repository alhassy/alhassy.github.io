;; Read a YAML file that defines input and expectations.
;; Evaluate the arithmetic expression in input.
;; Check if the result matches the expectations in both value and parity.
;; Recursively read all YAML files in a specified directory.
;;

(use-package yaml)

(defun read-yaml-file (file)
  "Read and parse a YAML FILE, returning its contents as a hash table."
  (with-temp-buffer
    (insert-file-contents file)
    (yaml-parse-string (buffer-string))))

(with-temp-file "example.yaml"
  (insert "
input: |-
  let x = 2
  let y = 3
  x + y
expectations:
  parity: odd
  value: 5
"))
(setq example (read-yaml-file "example.yaml"))

(require 'map)
(map-keys example)       ;; ⇒ (expectations input)
(map-elt example 'input) ;; ⇒ "2 + 2"
(map-nested-elt example '(expectations parity)) ;; ⇒ "odd"
(map-let (input expectations) example
  (map-let (parity value) expectations
    (format "Does  %s  equal  %s, which is %s?" (s-replace "\n" "; " input) value parity)))
;; ⇒ Does  let x = 2; let y = 3; x + y  equal  5, which is odd?

;; More tersely[?], we can specify “'key-name value-name” pairs using -let;
;; if value-name is the same as key-name, then we can omit it and it is inferred.
;; Below symbolic key 'input has no name after it, so a variable named “input” is created for us.
;; However, symbolic key 'expectations has a value _pattern_ whose keys have inferred value names.
(-let [(&hash 'input 'expectations (&hash 'value 'parity)) example]
  (format "Does  %s  equal  %s, which is %s?" (s-replace "\n" "; " input) value parity))

;; "Evaluate the JavaScript arithmetic EXPRESSION and return its result as a number."
;; Define a function to calculate both value and parity
(defun compute-values (input)
  "Calculate the arithmetic value and parity for the given INPUT string. Returns a hashmap."
  "Evaluate the JavaScript arithmetic EXPRESSION and return its result as a number."
  (let ((result
         (thread-last input shell-quote-argument (concat "node -p ") shell-command-to-string string-to-number)))
    (map-into (list 'value result                   ;; Construct the result as a list
                    'parity (if (ignore-errors (evenp result)) "even" "odd"))
              'hash-table
              )))  ;; Determine parity
;;
;; Example use
(compute-values (map-elt example 'input))

(require 'ert)

(defun make-ert-test-for-yaml-file (file)
  "ERT-compatible test for a YAML FILE based on 'input' and 'expectations'."
  ;; Create a separate ert test for each YAML file
  (let* ((data (read-yaml-file file))
         (input (gethash 'input data))
         (expected-results (gethash 'expectations data))
         (actual-results (compute-values input)))

    (eval `(ert-deftest ,(intern (format "e2e/%s" (file-name-base file))) ()
             ;; Iterate over each key in expected-results and create assertions
             (map-every-p (lambda (key expected-value)
                            (let ((actual-value (map-elt ,actual-results key 'not-found)))

                              (ert-info (`(lambda ()
                                            (when ,(and (stringp actual-value) (stringp expected-value))
                                              (insert-button "Show Diff"
                                                             'action (lambda (button)
                                                                       (show-string-diff ,expected-value ,actual-value))
                                                             'follow-link t
                                                             'help-echo "Click to see the diff."))
                                            (format " Test failed in file “%s” for key “%s”" ,,file (quote ,key))))
                                (should (equal actual-value expected-value)))))
                          ,expected-results)))))

;; Example use
(ert (make-ert-test-for-yaml-file  "example.yaml"))

(defun show-string-diff (string1 string2)
  "Show the difference between STRING1 and STRING2 using `diff` in Emacs."
  (interactive "sEnter first string: \nsEnter second string: ")
  (let ((temp-buffer1 (get-buffer-create "*Diff String 1*"))
        (temp-buffer2 (get-buffer-create "*Diff String 2*")))
    ;; Fill the first buffer with string1
    (with-current-buffer temp-buffer1
      (erase-buffer)
      (insert string1))

    ;; Fill the second buffer with string2
    (with-current-buffer temp-buffer2
      (erase-buffer)
      (insert string2))

    ;; Call the diff command on the two buffers
    (diff temp-buffer1 temp-buffer2)))

(cl-defun run-e2e-tests (&optional (directory "."))
  (interactive)
  "Run all e2e tests recursively in DIRECTORY."
  (loop for file in (directory-files-recursively directory "\\.yaml\\'")
        do (make-ert-test-for-yaml-file file))
  ;; Run all tests matching regex
  (ert "e2e/.*"))

(cl-defun update-e2e-tests ()
  (interactive)
  ;; update all tests
  (loop for file in (directory-files-recursively "." "\\.yaml\\'")
        for input = (map-elt (read-yaml-file file) 'input)
        ;; Creating an alist and not a hash-table so that ordering matters, for yaml encoding
        for actual = (list
                      (cons 'input input)
                      (cons 'expectations (map-into (compute-values input) 'alist)))
        do (with-temp-file file
             ;; (insert (yaml-encode actual)) ;; Nope, it does not honour new lines
             (insert (yaml-encode-alist actual))
             (-let [format-all-formatters '(("YAML" prettier))]
               (use-package yaml-mode) (yaml-mode) (use-package format-all) (format-all-buffer))
             )))


(defun yaml-encode-alist (alist &optional indent-level)
  "Encode an ALIST as a YAML-like string with multiline support.
Newlines within values are formatted using `|-' style.
Ensures a blank line before the `expectations' key if present.
INDENT-LEVEL specifies the current indentation level, defaulting to 0."
  (let ((indent-level (or indent-level 0))
        (first-key t))  ;; Track if it’s the first key
    (mapconcat
     (lambda (pair)
       (let* ((key (car pair))
              (value (cdr pair))
              (indent (make-string (* 2 indent-level) ? ))
              (sub-indent (make-string (* 2 (1+ indent-level)) ? ))
              ;; Insert a blank line before `expectations`
              (separator (if (and (eq key 'expectations) (not first-key))
                             "\n\n" "\n")))
         (setq first-key nil)  ;; No longer the first key after first iteration
         (concat separator
                 indent (symbol-name key) ": "
                 (cond
                  ;; Multiline strings: use `|-` style with additional indentation
                  ((and (stringp value) (string-match-p "\n" value))
                   (concat "|-\n"
                           sub-indent
                           (replace-regexp-in-string "\n" (concat "\n" sub-indent) value)))
                  ;; Nested alists: recursively call yaml-encode-alist
                  ((and (listp value) (listp (car value)))
                   (concat "\n" (yaml-encode-alist value (1+ indent-level))))
                  ;; Other values: simply output the value
                  (t
                   (prin1-to-string value))))))
     alist
     "")))
;; Example usage
(let ((example `((input . "let x = 2\nlet y = 3\nx + y + z")
                 (expectations . ((html . "<p>\nlet y = 3\nx + y + z</p>")
                                  (latex . "\\newline \n$let y = 3$ \n\\newline")
                                  (age . 5 ))))))
  (yaml-encode-alist example))
