#+begin_src emacs-lisp
;; When “C-x g” this shows me where the time goes.
;; Press “C-h e” and look for the items marked “!”, those took too much time.
;; I add those too my loop below ---except anything mentioning ‘stage’!
(setq magit-refresh-verbose  t)

(setq magit-status-sections-hook nil)
(loop for feature in '(magit-insert-unpushed-to-upstream-or-recent
                         ;; magit-insert-stashes
                         ;; magit-insert-untracked-files
                         ;; magit-insert-unpulled-from-upstream
                         ;; magit-insert-unpulled-from-pushremote
                         ;; magit-insert-unpushed-to-pushremote
                         ;; magit-insert-status-headers
                         ;; magit-insert-tags-header
                         magit-insert-unstaged-changes
                         magit-insert-merge-log
                         ;; magit-insert-rebase-sequence
                         ;; magit-insert-am-sequence
                         ;; magit-insert-sequencer-sequence
                         ;; magit-insert-bisect-output
                         ;; magit-insert-bisect-rest
                         ;; magit-insert-bisect-log
                         magit-insert-staged-changes
                         magit-todos--insert-todos)
do (add-hook 'magit-status-sections-hook feature))
#+end_src
