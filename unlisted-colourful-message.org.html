(defun go! ()
  "To actually see the colouring, run “M-x go!”

NOTE: Pressing “C-x C-e” after a “message” form will NOT show you the propertized text:
The message in the mode-line is getting immediately replaced by the return value. That's why you can't see it.
As such, using an “interactive” form will work, or you can simply pause:

   ;; Press C-x C-e and you'll actually see the colourful text
   (progn (message (propertize \"foo\" 'face '(:foreground \"red\"))) (sit-for 3))

Neither will:
→ (funcall-interactively 'go!)
→ (funcall-interactively  'execute-extended-command nil \"go!\")
"
  (interactive)
  ;; (message (propertize "text" 'face 'font-lock-warning-face))

  (message (concat  ;; "Click to see full details"
            (propertize "\n" 'face '(:height 0.4))
            (propertize "[Click]" 'face `(bold (foreground-color . "green"))) " To see detailed weather report\n"
            (propertize "[M-x my/weather-update]" 'face '(bold (foreground-color . "maroon"))) " To fetch latest weather data"
            (propertize "\n " 'face '(:height 0.5)))))


;; Press C-x C-e and you'll actually see the colourful text
(progn (message (propertize "foo" 'face '(:foreground "red"))) (sit-for 3))


;; Another example
(progn
  (message (propertize "HOLA" 'face `(:foreground "green" :slant italic :box t)))
  (sit-for 3))
#+end_src
